/* DUPLICATE MULTIPLE SELECTED ROOMS */ async function duplicateSelectedRooms() { const selectedCheckboxes = document.querySelectorAll('.row-checkbox:checked'); if (selectedCheckboxes.length === 0) { showAlert('Warning', 'Please select at least one room to duplicate using the checkbox!', 'warning'); return; } const selectedRows = Array.from(selectedCheckboxes).map(cb => cb.value); const { value: targetFloor } = await Swal.fire({ title: `Duplicate ${selectedRows.length} Room(s)`, text: 'Enter target floor name for the duplicated room(s):', input: 'text', inputPlaceholder: 'e.g., 2nd Floor / Floor 2', showCancelButton: true, confirmButtonText: 'Duplicate Rooms', confirmButtonColor: '#8b5cf6', cancelButtonColor: '#6b7280', inputValidator: (value) => { if (!value || !value.trim()) { return 'Target floor name cannot be empty!'; } } }); if (targetFloor) { const activeProject = getSelectedProject(); Swal.fire({ title: 'Duplicating Rooms...', allowOutsideClick: false, didOpen: () => { Swal.showLoading(); } }); try { const response = await fetch(SCRIPT_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "duplicateSelected", selectedRows: selectedRows, targetFloor: targetFloor.trim(), surveyorName: currentUser, userPin: currentPin, newProjectName: activeProject }) }); const result = await response.json(); if (result.status === "success") { showAlert('Success!', result.message, 'success'); fetchRoomData(); } else { showAlert('Duplicate Failed', result.message, 'error'); } } catch (err) { showAlert('Error', 'Connection error occurred during room duplication.', 'error'); } } }
